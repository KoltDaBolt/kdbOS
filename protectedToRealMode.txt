The following includes code demonstrating how to switch back to real mode after having switched to protected mode.

org 0x7C00
bits 16

entry:
    mov ax, 0
    mov ds, ax
    mov es, ax
    mov ss, ax
    mov sp, 0x7C00

    cli
    call enableA20
    call loadGDT
    mov eax, cr0
    or al, 1
    mov cr0, eax
    jmp dword 0x08:.pmode

.pmode:
    [bits 32]

    mov ax, 0x10
    mov ds, ax
    mov ss, ax

    mov esi, g_Hello
    mov edi, ScreenBuffer
    cld

.loop:
    lodsb
    or al, al
    jz .done

    mov [edi], al
    inc edi

    inc edi
    jmp .loop

.done:
    jmp 0x18:.pmode16

.pmode16:
    [bits 16]

    mov eax, cr0
    and al, ~1
    mov cr0, eax

    jmp word 0x00:.rmode

.rmode:
    mov ax, 0
    mov ds, ax
    mov ss, ax

    sti

    mov si, g_HelloR

.rloop:
    lodsb
    or al, al
    jz .rdone
    mov ah, 0x0E
    int 0x10
    jmp .rloop

.rdone:

.halt:
    jmp .halt
